cmake_minimum_required(VERSION 3.19)
project(vinyl_artist_artwork_resolver LANGUAGES CXX)
enable_testing()
add_executable(test_artist_artwork ArtistArtworkResolver.cpp test_artist_artwork.cpp)
target_compile_features(test_artist_artwork PRIVATE cxx_std_17)
set_target_properties(test_artist_artwork PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(test_artist_artwork PRIVATE /W4 /WX)
else()
    target_compile_options(test_artist_artwork PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME artist_artwork_deterministic COMMAND test_artist_artwork)
