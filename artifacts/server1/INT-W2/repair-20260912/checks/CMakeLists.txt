cmake_minimum_required(VERSION 3.24)
project(server1_w2_link_proof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 CONFIG REQUIRED COMPONENTS Core Network)
get_filename_component(SERVER1_REPO "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(W2_BUILD_DIR "" CACHE PATH "Fresh assembled Server 1.0 build directory")
set(W2_LIBRARIES server1_k00_policy server1_k01_metadata server1_k13_settings_cache
    server1_h00_http server1_m00_process)
foreach(target IN LISTS W2_LIBRARIES)
  set(library "${W2_BUILD_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}${target}${CMAKE_STATIC_LIBRARY_SUFFIX}")
  if(NOT EXISTS "${library}")
    message(FATAL_ERROR "Build the assembled tree first; missing ${library}")
  endif()
  add_library(${target} STATIC IMPORTED)
  set_target_properties(${target} PROPERTIES IMPORTED_LOCATION "${library}")
endforeach()
add_executable(combined_w2 combined_w2.cpp)
target_include_directories(combined_w2 PRIVATE "${SERVER1_REPO}/native/colosseum_server_v1/include")
target_link_libraries(combined_w2 PRIVATE ${W2_LIBRARIES} Qt6::Core Qt6::Network)
add_executable(k13a_assembled_probe
    "${SERVER1_REPO}/artifacts/server1/K13/K13-A/probe-src/settings_store_probe.cpp")
target_include_directories(k13a_assembled_probe PRIVATE "${SERVER1_REPO}/native/colosseum_server_v1/include")
target_link_libraries(k13a_assembled_probe PRIVATE server1_k13_settings_cache server1_k00_policy Qt6::Core)
