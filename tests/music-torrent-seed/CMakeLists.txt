cmake_minimum_required(VERSION 3.20)
project(MusicTorrentSeed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(LibtorrentRasterbar CONFIG QUIET)
add_executable(music_seed ../vinyl-live/music_seed.cpp)
target_link_libraries(music_seed PRIVATE Qt6::Core Qt6::Network)
if(TARGET LibtorrentRasterbar::torrent-rasterbar)
  target_link_libraries(music_seed PRIVATE LibtorrentRasterbar::torrent-rasterbar)
elseif(WIN32)
  set(LIBTORRENT_ROOT "" CACHE PATH "libtorrent installation")
  set(BOOST_ROOT "" CACHE PATH "Boost matching libtorrent ABI")
  set(OPENSSL_ROOT "" CACHE PATH "OpenSSL installation")
  if(NOT LIBTORRENT_ROOT OR NOT BOOST_ROOT OR NOT OPENSSL_ROOT)
    message(FATAL_ERROR "Set LIBTORRENT_ROOT, BOOST_ROOT and OPENSSL_ROOT to matching build dependencies")
  endif()
  target_include_directories(music_seed PRIVATE "${LIBTORRENT_ROOT}/include" "${BOOST_ROOT}" "${OPENSSL_ROOT}/include")
  target_compile_definitions(music_seed PRIVATE TORRENT_USE_OPENSSL BOOST_ALL_NO_LIB _WIN32_WINNT=0x0A00)
  target_link_libraries(music_seed PRIVATE "${LIBTORRENT_ROOT}/lib/torrent-rasterbar.lib" "${OPENSSL_ROOT}/lib/libssl.lib" "${OPENSSL_ROOT}/lib/libcrypto.lib" ws2_32 mswsock crypt32 iphlpapi bcrypt)
else()
  message(FATAL_ERROR "LibtorrentRasterbar required")
endif()
