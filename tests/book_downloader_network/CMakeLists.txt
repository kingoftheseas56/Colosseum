cmake_minimum_required(VERSION 3.21)
project(book_downloader_network LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
set(BOOK_DOWNLOADER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../native/engine" CACHE PATH "Production BookDownloader source directory (override for red baseline)")
add_executable(book_downloader_network book_downloader_network.cpp "${BOOK_DOWNLOADER_SOURCE_DIR}/BookDownloader.cpp" "${BOOK_DOWNLOADER_SOURCE_DIR}/BookDownloader.h")
target_include_directories(book_downloader_network PRIVATE "${BOOK_DOWNLOADER_SOURCE_DIR}")
target_link_libraries(book_downloader_network PRIVATE Qt6::Core Qt6::Network Qt6::Test)
if(MSVC)
 target_compile_options(book_downloader_network PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME book_downloader_network COMMAND book_downloader_network)
set_tests_properties(book_downloader_network PROPERTIES TIMEOUT 90)
