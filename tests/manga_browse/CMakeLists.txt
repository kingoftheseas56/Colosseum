cmake_minimum_required(VERSION 3.21)
project(MangaBrowseTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml Quick QuickTest)
add_executable(manga_browse_quicktest ../manga_browse_quicktest_main.cpp)
target_link_libraries(manga_browse_quicktest PRIVATE Qt6::Core Qt6::Qml Qt6::Quick Qt6::QuickTest)
enable_testing()
add_test(NAME manga_browse_qml COMMAND manga_browse_quicktest -input ${CMAKE_CURRENT_SOURCE_DIR}/../qml/manga_browse)
set_tests_properties(manga_browse_qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
