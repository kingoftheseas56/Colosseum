cmake_minimum_required(VERSION 3.24)
project(FeriaFoundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Qml Quick QuickTest Test Network Sql WebEngineQuick)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_library(foundation_owners STATIC
    "${ROOT}/native/SessionStore.h"
    "${ROOT}/native/feria/FeriaAccountStore.cpp"
    "${ROOT}/native/feria/FeriaBrowserPolicy.cpp"
    "${ROOT}/native/feria/app-state/PorticoAppStateBridge.cpp")
target_include_directories(foundation_owners PUBLIC "${ROOT}/native" "${ROOT}/native/feria")
target_link_libraries(foundation_owners PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Sql)
add_executable(feria_foundation_policy_test "${ROOT}/tests/feria_foundation_policy_test.cpp")
target_link_libraries(feria_foundation_policy_test PRIVATE foundation_owners Qt6::Test)
add_executable(feria_foundation_runtime "${ROOT}/tests/feria_foundation_runtime.cpp")
target_link_libraries(feria_foundation_runtime PRIVATE foundation_owners Qt6::QuickTest Qt6::WebEngineQuick Qt6::Network)
add_executable(feria_session_ownership_test "${ROOT}/tests/feria_session_ownership_test.cpp")
target_link_libraries(feria_session_ownership_test PRIVATE foundation_owners Qt6::Test)
enable_testing()
add_test(NAME foundation_policy COMMAND feria_foundation_policy_test)
add_test(NAME session_ownership COMMAND feria_session_ownership_test)
add_executable(feria_hosting_state_test "${ROOT}/tests/feria_hosting_state_test.cpp")
target_link_libraries(feria_hosting_state_test PRIVATE foundation_owners Qt6::Test)
add_test(NAME hosting_state COMMAND feria_hosting_state_test)
add_executable(feria_hosting_geometry_test "${ROOT}/tests/feria_hosting_geometry_test.cpp")
target_link_libraries(feria_hosting_geometry_test PRIVATE foundation_owners Qt6::Test)
add_test(NAME hosting_geometry COMMAND feria_hosting_geometry_test)
