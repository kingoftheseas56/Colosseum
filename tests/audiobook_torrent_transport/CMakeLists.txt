cmake_minimum_required(VERSION 3.21)
project(audiobook_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
add_executable(audiobook_transport_test test.cpp stream_stub.cpp ../../native/engine/AudiobookDownloader.cpp ../../native/engine/AudiobookDownloader.h ../../native/AudioPairingStore.h ../../native/player/streamserver.h)
target_sources(audiobook_transport_test PRIVATE ../../native/engine/AudiobookTorrentTransport.cpp ../../native/engine/AudiobookTorrentTransport.h ../../native/engine/AudiobookTorrentEngineAdapter.h)
target_link_libraries(audiobook_transport_test PRIVATE Qt6::Core Qt6::Network Qt6::Test)
if(MSVC)
 target_compile_options(audiobook_transport_test PRIVATE /utf-8)
endif()

# Compile the real concrete forwarding adapter without starting any engine.
add_library(audiobook_adapter_compile OBJECT ../../native/engine/AudiobookTorrentEngineAdapter.cpp)
target_link_libraries(audiobook_adapter_compile PRIVATE Qt6::Core Qt6::Network)
if(MSVC)
 target_compile_options(audiobook_adapter_compile PRIVATE /utf-8)
endif()
option(AUDIOBOOK_LOOPBACK_FIXTURE "Build isolated real libtorrent fixture with installed host dependencies" OFF)
if(AUDIOBOOK_LOOPBACK_FIXTURE)
 find_package(LibtorrentRasterbar CONFIG REQUIRED)
 add_executable(audiobook_loopback_fixture loopback_fixture.cpp ../../native/engine/AudiobookTorrentTransport.cpp ../../native/engine/AudiobookTorrentTransport.h ../../native/engine/AudiobookTorrentEngineAdapter.h)
 target_link_libraries(audiobook_loopback_fixture PRIVATE Qt6::Core Qt6::Network Qt6::Test LibtorrentRasterbar::torrent-rasterbar)
 if(MSVC)
  target_compile_options(audiobook_loopback_fixture PRIVATE /utf-8)
 endif()
 target_compile_definitions(audiobook_loopback_fixture PRIVATE BOOST_ALL_NO_LIB)
endif()
if(AUDIOBOOK_LOOPBACK_FIXTURE)
 add_executable(audiobook_owned_removal_contract owned_removal_contract.cpp)
 target_compile_definitions(audiobook_owned_removal_contract PRIVATE HAS_LIBTORRENT BOOST_ALL_NO_LIB)
 if(MSVC)
  target_compile_options(audiobook_owned_removal_contract PRIVATE /utf-8)
 endif()
 target_link_libraries(audiobook_owned_removal_contract PRIVATE Qt6::Core Qt6::Test LibtorrentRasterbar::torrent-rasterbar)
 # Compile the actual engine extension; do not construct/start its public-discovery session.
 add_library(audiobook_real_engine_compile OBJECT ../../native/torrent/engine/TorrentEngine.cpp)
 target_compile_definitions(audiobook_real_engine_compile PRIVATE HAS_LIBTORRENT BOOST_ALL_NO_LIB)
 if(MSVC)
  target_compile_options(audiobook_real_engine_compile PRIVATE /utf-8)
 endif()
 target_link_libraries(audiobook_real_engine_compile PRIVATE Qt6::Core Qt6::Network LibtorrentRasterbar::torrent-rasterbar)
endif()


if(WIN32 AND AUDIOBOOK_LOOPBACK_FIXTURE)
 foreach(target audiobook_loopback_fixture audiobook_owned_removal_contract audiobook_real_engine_compile)
  target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0A00)
 endforeach()
endif()
