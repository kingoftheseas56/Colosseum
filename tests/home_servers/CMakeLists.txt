cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(HomeServerClientsTests LANGUAGES CXX)
  include(CTest)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
if(NOT TARGET colosseum_account_credentials)
  add_subdirectory(../../native/account/credentials credentials)
endif()
if(NOT TARGET colosseum_home_servers)
  add_subdirectory(../../native/homeserver homeserver)
endif()
add_executable(tst_home_server_clients tst_home_server_clients.cpp)
set_target_properties(tst_home_server_clients PROPERTIES AUTOMOC ON)
target_compile_features(tst_home_server_clients PRIVATE cxx_std_17)
target_link_libraries(tst_home_server_clients PRIVATE colosseum_home_servers Qt6::Test)
target_compile_definitions(tst_home_server_clients PRIVATE HOME_SERVER_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_test(NAME home_server_clients COMMAND tst_home_server_clients)

foreach(surface store service)
