cmake_minimum_required(VERSION 3.21)
project(feria_html_mpv_composition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fPIC) # Avoid Qt global COPY relocations across compatible runtime minors.
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick WebEngineQuick WebChannel Test)
find_package(MpvQt REQUIRED)
enable_testing()
qt_add_executable(session_tests tst_session.cpp)
target_link_libraries(session_tests PRIVATE Qt6::Test)
add_test(NAME session_tests COMMAND session_tests)
qt_add_executable(feria_html_mpv main.cpp bridge.cpp bridge.h
    ../../native/player/mpvitem.cpp ../../native/player/mpvitem.h)
target_include_directories(feria_html_mpv PRIVATE ../../native/player)
qt_add_resources(feria_html_mpv assets PREFIX "/" FILES Main.qml index.html adapter.js selftest.js)
target_link_libraries(feria_html_mpv PRIVATE Qt6::Quick Qt6::WebEngineQuick Qt6::WebChannel Qt6::Test MpvQt::MpvQt)
