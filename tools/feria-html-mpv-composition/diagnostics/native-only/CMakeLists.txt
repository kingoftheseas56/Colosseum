cmake_minimum_required(VERSION 3.21)
project(feria_native_only_baseline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fPIC)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick)
find_package(MpvQt REQUIRED)
qt_add_executable(native_only_baseline main.cpp
    ../../../../native/player/mpvitem.cpp ../../../../native/player/mpvitem.h)
target_include_directories(native_only_baseline PRIVATE ../../../../native/player)
qt_add_resources(native_only_baseline assets PREFIX "/" FILES Main.qml)
target_link_libraries(native_only_baseline PRIVATE Qt6::Quick MpvQt::MpvQt)
