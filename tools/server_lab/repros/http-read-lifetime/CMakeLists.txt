cmake_minimum_required(VERSION 3.24)
project(server1_read_lifetime_repro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Network)
if(NOT ROOT)
 message(FATAL_ERROR "Specify exact HTTP source ROOT")
endif()
add_executable(repro repro.cpp "${ROOT}/src/http/Connection.cpp" "${ROOT}/src/http/Router.cpp" "${ROOT}/src/http/RequestParsing.cpp")
target_include_directories(repro PRIVATE "${ROOT}/include")
target_link_libraries(repro PRIVATE Qt6::Core Qt6::Network)
